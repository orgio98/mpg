    function splitActiveRangeAtPlayhead() {
      const clip = getSelectedClip();
      const range = getActiveRange(clip);
      if (!clip || !range) return;

      const splitPoint = Number((el.previewVideo.currentTime || 0).toFixed(3));
      const minStart = range.start + MIN_SEG_SEC;
      const maxEnd = range.end - MIN_SEG_SEC;

      if (minStart >= maxEnd) {
        alert('분할하기에 활성 구간 길이가 너무 짧습니다 (최소 0.10초 이상 필요).');
        return;
      }

      if (splitPoint <= minStart || splitPoint >= maxEnd) {
        alert(`재생 헤드를 활성 구간 내부(${formatPreciseTime(minStart)} ~ ${formatPreciseTime(maxEnd)})에 위치시킨 뒤 분할해 주세요.`);
        return;
      }

      const idx = clip.ranges.findIndex(r => r.id === range.id);
      const firstPart = { id: range.id, start: range.start, end: splitPoint, enabled: range.enabled };
      const secondPart = { id: uid(), start: splitPoint, end: range.end, enabled: range.enabled };

      clip.ranges.splice(idx, 1, firstPart, secondPart);
      clip.activeRangeId = secondPart.id;

      appendLog(`[분할] ${clip.name} · ${formatPreciseTime(splitPoint)} 지점에서 구간을 2개로 분할했습니다.`);
      rebuildTimelineStructure(true);
      renderRanges();
      renderClips();
    }

    async function extractThumbnails(file, duration, count = 16) {
      const thumbs = [];
      if (!duration || duration <= 0) return thumbs;

      const url = URL.createObjectURL(file);
      const v = document.createElement('video');
      v.src = url;
      v.muted = true;
      v.playsInline = true;

      await new Promise(res => {
        v.onloadedmetadata = res;
        v.onerror = res;
      });

      const canvas = document.createElement('canvas');
      canvas.width = 120;
      canvas.height = 68;
      const ctx = canvas.getContext('2d');

      for (let i = 0; i < count; i++) {
        const t = Math.min(duration - 0.05, Math.max(0, (duration * (i + 0.5)) / count));
        v.currentTime = t;
        await new Promise(res => {
          const timer = setTimeout(res, 250);
          v.onseeked = () => {
            clearTimeout(timer);
            res();
          };
        });
        try {
          ctx.drawImage(v, 0, 0, canvas.width, canvas.height);
          thumbs.push(canvas.toDataURL('image/jpeg', 0.55));
        } catch (_) {}
      }

      URL.revokeObjectURL(url);
      return thumbs;
    }

    // 눈금자는 클립 변경 또는 줌 변경 시에만 재생성 (드래그 중 재렌더링 방지)
    function renderRuler(duration) {
      el.timelineRuler.innerHTML = '';
      if (!duration || duration <= 0) return;

      const step = zoomLevel >= 6 ? 0.5 : zoomLevel >= 3 ? 1 : zoomLevel >= 1.5 ? 2 : 5;
      const minorStep = step / 5;
      const frag = document.createDocumentFragment();

      for (let t = 0; t <= duration + 0.001; t += minorStep) {
        const pct = Math.min(100, (t / duration) * 100);
        const isMajor = Math.abs(Math.round(t / step) * step - t) < 0.01;

        const tick = document.createElement('div');
        tick.className = `ruler-tick ${isMajor ? 'major' : 'minor'}`;
        tick.style.left = `${pct}%`;
        frag.appendChild(tick);

        if (isMajor) {
          const label = document.createElement('div');
          label.className = 'ruler-label';
          label.style.left = `${pct}%`;
          label.textContent = formatPreciseTime(t).slice(0, -1);
          frag.appendChild(label);
        }
      }
      el.timelineRuler.appendChild(frag);
    }

    function rebuildTimelineStructure(autoScrollToPlayhead = false) {
      const clip = getSelectedClip();
      const activeRange = getActiveRange(clip);

      el.filmstripTrack.querySelectorAll('.timeline-ghost-range').forEach(n => n.remove());

      if (!clip || clip.duration <= 0 || !activeRange) {
        el.vlloEditor.classList.add('disabled');
        el.rangeSummary.textContent = '00:00.00 ~ 00:00.00';
        el.rangeDuration.textContent = '0.00초';
        el.startPreciseLabel.textContent = '00:00.00';
        el.endPreciseLabel.textContent = '00:00.00';
        el.timecodeHud.textContent = '00:00.00 / 00:00.00';
        return;
      }

      el.vlloEditor.classList.remove('disabled');
      el.timelineCanvas.style.width = `${zoomLevel * 100}%`;
      renderRuler(clip.duration);

      clip.ranges.forEach(r => {
        if (r.id === activeRange.id) return;
        const gStart = (r.start / clip.duration) * 100;
        const gWidth = Math.max(0.3, ((r.end - r.start) / clip.duration) * 100);
        const ghost = document.createElement('div');
        ghost.className = 'timeline-ghost-range';
        ghost.style.left = `${gStart}%`;
        ghost.style.width = `${gWidth}%`;
        el.filmstripTrack.appendChild(ghost);
      });

      updateTimelinePositions(autoScrollToPlayhead);
    }

    // 초경량 위치·텍스트 업데이트 (드래그 중 60fps 실행)
    function updateTimelinePositions(autoScrollToPlayhead = false) {
      const clip = getSelectedClip();
      const activeRange = getActiveRange(clip);
      if (!clip || clip.duration <= 0 || !activeRange) return;

      const startPct = (activeRange.start / clip.duration) * 100;
      const endPct = (activeRange.end / clip.duration) * 100;

      el.handleStart.style.left = `${startPct}%`;
      el.handleEnd.style.left = `${endPct}%`;
      el.timelineRange.style.left = `${startPct}%`;
      el.timelineRange.style.width = `${Math.max(0, endPct - startPct)}%`;

      const startStr = formatPreciseTime(activeRange.start);
      const endStr = formatPreciseTime(activeRange.end);
      el.tooltipStart.textContent = startStr;
      el.tooltipEnd.textContent = endStr;
      el.startPreciseLabel.textContent = startStr;
      el.endPreciseLabel.textContent = endStr;

      el.rangeSummary.textContent = `${startStr} ~ ${endStr}`;
      el.rangeDuration.textContent = `${Math.max(0, activeRange.end - activeRange.start).toFixed(2)}초`;

      const displayTime = scrubTime !== null ? scrubTime : (el.previewVideo.currentTime || 0);
      const playPct = Math.min(100, Math.max(0, (displayTime / clip.duration) * 100));
      el.timelinePlayhead.style.left = `${playPct}%`;
      el.timecodeHud.textContent = `${formatPreciseTime(displayTime)} / ${formatPreciseTime(clip.duration)}`;

      if (autoScrollToPlayhead && zoomLevel > 1) {
        const canvasW = el.timelineCanvas.offsetWidth;
        const viewW = el.timelineViewport.clientWidth;
        const playX = (playPct / 100) * canvasW;
        el.timelineViewport.scrollLeft = Math.max(0, playX - viewW * 0.5);
      }
    }

    function nudgeActiveRange(target, delta) {
      const clip = getSelectedClip();
      const range = getActiveRange(clip);
      if (!clip || !range) return;

      stopSequencePreview(true);
      const minGap = Math.min(MIN_SEG_SEC, clip.duration);

      if (target === 'start') {
        const next = Math.max(0, Math.min(range.start + delta, range.end - minGap));
        range.start = Number(next.toFixed(3));
        el.previewVideo.currentTime = range.start;
      } else {
        const next = Math.min(clip.duration, Math.max(range.end + delta, range.start + minGap));
        range.end = Number(next.toFixed(3));
        el.previewVideo.currentTime = range.end;
      }

      updateTimelinePositions(true);
      renderRanges();
      renderClips();
    }

    function nudgePlayhead(delta) {
      const clip = getSelectedClip();
      if (!clip) return;
      stopSequencePreview(true);
      const next = Math.max(0, Math.min(clip.duration, (el.previewVideo.currentTime || 0) + delta));
      el.previewVideo.currentTime = Number(next.toFixed(3));
      updateTimelinePositions(true);
    }

    function renderRanges() {
      const clip = getSelectedClip();
      el.rangeList.innerHTML = '';

      if (!clip) {
        el.rangeCount.textContent = '0';
        el.checkedCount.textContent = '0';
        el.seqTotalDuration.textContent = '0.00초';
        updateButtons();
        return;
      }

      const checkedRanges = clip.ranges.filter(r => r.enabled);
      el.rangeCount.textContent = clip.ranges.length;
      el.checkedCount.textContent = checkedRanges.length;
      el.seqTotalDuration.textContent = `${getCheckedDuration(clip).toFixed(2)}초`;

      clip.ranges.forEach((r, idx) => {
        const li = document.createElement('li');
        const isActive = r.id === clip.activeRangeId;
        li.className = `range-item ${isActive ? 'active' : ''} ${!r.enabled ? 'disabled-range' : ''}`;
        li.onclick = () => {
          clip.activeRangeId = r.id;
          el.previewVideo.currentTime = r.start;
          rebuildTimelineStructure(true);
          renderRanges();
        };

        const dur = Math.max(0, r.end - r.start).toFixed(2);
        li.innerHTML = `
          <div class="range-item-left">
            <input type="checkbox" ${r.enabled ? 'checked' : ''} title="연속 미리보기 및 병합에 포함" />
            <div class="range-item-info">
              <strong>구간 #${idx + 1}</strong> · ${formatPreciseTime(r.start)} ~ ${formatPreciseTime(r.end)} (${dur}초)
            </div>
          </div>
          <div class="range-item-actions" onclick="event.stopPropagation()">
            <button type="button" title="위로" ${idx === 0 ? 'disabled' : ''} data-act="up">▲</button>
            <button type="button" title="아래로" ${idx === clip.ranges.length - 1 ? 'disabled' : ''} data-act="down">▼</button>
            <button type="button" class="danger" title="삭제" ${clip.ranges.length <= 1 ? 'disabled' : ''} data-act="del">✕</button>
          </div>
        `;

        const checkbox = li.querySelector('input[type="checkbox"]');
        checkbox.onclick = e => {
          e.stopPropagation();
          r.enabled = checkbox.checked;
          renderRanges();
          renderClips();
        };

        li.querySelector('[data-act="up"]').onclick = () => {
          const [item] = clip.ranges.splice(idx, 1);
          clip.ranges.splice(idx - 1, 0, item);
          renderRanges();
          renderClips();
        };

        li.querySelector('[data-act="down"]').onclick = () => {
          const [item] = clip.ranges.splice(idx, 1);
          clip.ranges.splice(idx + 1, 0, item);
          renderRanges();
          renderClips();
        };

        li.querySelector('[data-act="del"]').onclick = () => {
          if (clip.ranges.length <= 1) return;
          clip.ranges.splice(idx, 1);
          if (clip.activeRangeId === r.id) {
            clip.activeRangeId = clip.ranges[0].id;
          }
          rebuildTimelineStructure();
          renderRanges();
          renderClips();
        };

        el.rangeList.appendChild(li);
      });

      updateButtons();
    }

    function setupTimelineDrag() {
      const startDrag = (type, e) => {
        const clip = getSelectedClip();
        const range = getActiveRange(clip);
        if (!clip || !range) return;
        e.stopPropagation();
        e.preventDefault();

        stopSequencePreview(true);
        if (e.target && e.target.setPointerCapture) {
          try { e.target.setPointerCapture(e.pointerId); } catch (_) {}
        }

        const trackRect = el.filmstripTrack.getBoundingClientRect();
        dragState = {
          type,
          pointerId: e.pointerId,
          captureEl: e.target,
          startX: e.clientX,
          trackLeft: trackRect.left,
          trackWidth: trackRect.width,
          initStart: range.start,
          initEnd: range.end
        };

        if (type === 'start') el.handleStart.classList.add('dragging');
        if (type === 'end') el.handleEnd.classList.add('dragging');
        if (type === 'scrub') {
          const ratio = Math.min(1, Math.max(0, (e.clientX - trackRect.left) / trackRect.width));
          scrubTime = Number((ratio * clip.duration).toFixed(3));
          requestSmoothVideoSeek(scrubTime, true);
          updateTimelinePositions();
        }
      };

      el.handleStart.addEventListener('pointerdown', e => startDrag('start', e));
      el.handleEnd.addEventListener('pointerdown', e => startDrag('end', e));
      el.timelineRange.addEventListener('pointerdown', e => startDrag('range', e));

      // 눈금자 드래그 시 플레이헤드 부드럽게 스크러빙
      el.timelineRuler.addEventListener('pointerdown', e => startDrag('scrub', e));

      // 필름스트립 트랙 빈 영역 클릭/드래그 시에도 스크러빙
      el.filmstripTrack.addEventListener('pointerdown', e => {
        if (e.target === el.filmstripTrack) {
          startDrag('scrub', e);
        }
      });

      window.addEventListener('pointermove', e => {
        if (!dragState) return;
        const clip = getSelectedClip();
        const range = getActiveRange(clip);
        if (!clip || !range) return;

        const width = Math.max(1, dragState.trackWidth);
        const precisionFactor = e.shiftKey ? 0.2 : 1;
        const minGap = Math.min(MIN_SEG_SEC, clip.duration);

        if (dragState.type === 'scrub') {
          const ratio = Math.min(1, Math.max(0, (e.clientX - dragState.trackLeft) / width));
          scrubTime = Number((ratio * clip.duration).toFixed(3));
          requestSmoothVideoSeek(scrubTime, false);
        } else {
          const deltaSec = (((e.clientX - dragState.startX) / width) * clip.duration) * precisionFactor;

          if (dragState.type === 'start') {
            const nextStart = Math.max(0, Math.min(dragState.initStart + deltaSec, range.end - minGap));
            range.start = Number(nextStart.toFixed(3));
            scrubTime = range.start;
            requestSmoothVideoSeek(range.start, false);
          } else if (dragState.type === 'end') {
            const nextEnd = Math.min(clip.duration, Math.max(dragState.initEnd + deltaSec, range.start + minGap));
            range.end = Number(nextEnd.toFixed(3));
            scrubTime = range.end;
            requestSmoothVideoSeek(range.end, false);
          } else if (dragState.type === 'range') {
            const span = dragState.initEnd - dragState.initStart;
            let nextStart = dragState.initStart + deltaSec;
            if (nextStart < 0) nextStart = 0;
            if (nextStart + span > clip.duration) nextStart = clip.duration - span;
            range.start = Number(nextStart.toFixed(3));
            range.end = Number((nextStart + span).toFixed(3));
            scrubTime = range.start;
            requestSmoothVideoSeek(range.start, false);
          }
        }

        if (!rafPending) {
          rafPending = true;
          requestAnimationFrame(() => {
            rafPending = false;
            updateTimelinePositions();
          });
        }
      });

      window.addEventListener('pointerup', () => {
        if (!dragState) return;
        if (dragState.captureEl && dragState.captureEl.releasePointerCapture) {
          try { dragState.captureEl.releasePointerCapture(dragState.pointerId); } catch (_) {}
        }
        el.handleStart.classList.remove('dragging');
        el.handleEnd.classList.remove('dragging');

        // 손을 떼는 순간 보류된 최종 위치로 정확히 비디오 프레임 확정
        if (pendingSeekTime !== null) {
          el.previewVideo.currentTime = pendingSeekTime;
          pendingSeekTime = null;
        }
        const wasScrub = dragState.type === 'scrub';
        dragState = null;
        scrubTime = null;

        updateTimelinePositions();
        if (!wasScrub) {
          renderRanges();
          renderClips();
        }
      });

      el.previewVideo.addEventListener('timeupdate', () => {
        if (dragState) return;
        updateTimelinePositions();
      });

      el.zoomSlider.addEventListener('input', () => {
        zoomLevel = Number(el.zoomSlider.value);
        el.zoomLabel.textContent = `${zoomLevel.toFixed(1)}x`;
        rebuildTimelineStructure(true);
      });

      el.timelineViewport.addEventListener('wheel', e => {
        if (!e.ctrlKey && !e.metaKey) return;
        e.preventDefault();
        const nextZoom = Math.min(10, Math.max(1, zoomLevel + (e.deltaY < 0 ? 0.5 : -0.5)));
        zoomLevel = nextZoom;
        el.zoomSlider.value = zoomLevel;
        el.zoomLabel.textContent = `${zoomLevel.toFixed(1)}x`;
        rebuildTimelineStructure(true);
      }, { passive: false });

      document.querySelectorAll('[data-jog]').forEach(btn => {
        btn.addEventListener('click', () => {
          nudgeActiveRange(btn.dataset.jog, Number(btn.dataset.delta));
        });
      });

      el.headPrevSec.onclick = () => nudgePlayhead(-0.5);
      el.headPrevFrame.onclick = () => nudgePlayhead(-FRAME_SEC);
      el.headNextFrame.onclick = () => nudgePlayhead(FRAME_SEC);
      el.headNextSec.onclick = () => nudgePlayhead(0.5);
      el.splitRangeBtn.onclick = splitActiveRangeAtPlayhead;
      el.stopSeqBtn.onclick = () => stopSequencePreview(true);
      el.saveAsPickerBtn.onclick = saveResultToChosenLocation;

      window.addEventListener('keydown', e => {
        if (['INPUT', 'SELECT', 'TEXTAREA'].includes(document.activeElement?.tagName)) return;
        const clip = getSelectedClip();
        if (!clip) return;

        if (e.key === 'ArrowLeft') {
          e.preventDefault();
          nudgePlayhead(e.shiftKey ? -0.5 : -FRAME_SEC);
        } else if (e.key === 'ArrowRight') {
          e.preventDefault();
          nudgePlayhead(e.shiftKey ? 0.5 : FRAME_SEC);
        } else if (e.key.toLowerCase() === 'i') {
          e.preventDefault();
          el.setStartBtn.click();
        } else if (e.key.toLowerCase() === 'o') {
          e.preventDefault();
          el.setEndBtn.click();
        } else if (e.key.toLowerCase() === 's') {
          e.preventDefault();
          splitActiveRangeAtPlayhead();
        }
      });
    }

    async function ensureFFmpeg() {
      if (ffmpegLoaded) return;
      el.engineStatus.textContent = 'FFmpeg 코어 다운로드 중...';
      appendLog('[시스템] FFmpeg ESM Worker 및 코어를 불러오는 중입니다 (최초 1회 약 25MB)...');

      ffmpeg = new FFmpeg();
      ffmpeg.on('log', ({ message }) => appendLog(message));
      ffmpeg.on('progress', ({ progress }) => {
        if (Number.isFinite(progress)) {
          el.progressBar.value = Math.min(100, Math.max(0, Math.round(progress * 100)));
        }
      });

      const ffmpegEsmBase = 'https://cdn.jsdelivr.net/npm/@ffmpeg/ffmpeg@0.12.10/dist/esm';
      const coreEsmBase = 'https://cdn.jsdelivr.net/npm/@ffmpeg/core@0.12.6/dist/esm';

      const workerResp = await fetch(`${ffmpegEsmBase}/worker.js`);
      let workerCode = await workerResp.text();
      workerCode = workerCode
        .replace(/from\s+["']\.\/const\.js["']/g, `from "${ffmpegEsmBase}/const.js"`)
        .replace(/from\s+["']\.\/errors\.js["']/g, `from "${ffmpegEsmBase}/errors.js"`);
      const classWorkerURL = URL.createObjectURL(new Blob([workerCode], { type: 'text/javascript' }));

      await ffmpeg.load({
        classWorkerURL,
        coreURL: await toBlobURL(`${coreEsmBase}/ffmpeg-core.js`, 'text/javascript'),
        wasmURL: await toBlobURL(`${coreEsmBase}/ffmpeg-core.wasm`, 'application/wasm')
      });

      ffmpegLoaded = true;
      el.engineStatus.textContent = 'FFmpeg 준비 완료 (로컬 처리)';
      appendLog('[시스템] FFmpeg 엔진 초기화 완료.');
    }

    function probeVideoFile(file) {
      return new Promise((resolve, reject) => {
        const url = URL.createObjectURL(file);
        const tempVideo = document.createElement('video');
        tempVideo.preload = 'metadata';
        tempVideo.src = url;
        tempVideo.onloadedmetadata = () => {
          resolve({
            url,
            duration: Number(tempVideo.duration.toFixed(3)),
            width: tempVideo.videoWidth,
            height: tempVideo.videoHeight
          });
        };
        tempVideo.onerror = () => {
          URL.revokeObjectURL(url);
          reject(new Error(`메타데이터를 읽을 수 없습니다: ${file.name}`));
        };
      });
    }

    async function addFiles(fileList) {
      const files = Array.from(fileList).filter(f => f.type === 'video/mp4' || f.name.toLowerCase().endsWith('.mp4'));
      if (!files.length) {
        alert('MP4 형식의 파일만 추가할 수 있습니다.');
        return;
      }

      for (const file of files) {
        try {
          const meta = await probeVideoFile(file);
          const initialRangeId = uid();
          const newClip = {
            id: uid(),
            file,
            name: file.name,
            url: meta.url,
            duration: meta.duration,
            width: meta.width,
            height: meta.height,
            thumbs: [],
            ranges: [{ id: initialRangeId, start: 0, end: meta.duration, enabled: true }],
            activeRangeId: initialRangeId
          };
          clips.push(newClip);
          if (!selectedClipId) selectedClipId = newClip.id;

          extractThumbnails(file, meta.duration, 18).then(thumbs => {
            newClip.thumbs = thumbs;
            if (selectedClipId === newClip.id) renderFilmstrip(newClip);
          });
        } catch (err) {
          appendLog(`[오류] ${err.message}`);
        }
      }

      renderClips();
      selectClip(selectedClipId);
    }

    function selectClip(id, cancelSequence = true) {
      if (cancelSequence) {
        stopSequencePreview(true);
      }

      selectedClipId = id;
      const clip = getSelectedClip();

      if (previewStopHandler) {
        el.previewVideo.removeEventListener('timeupdate', previewStopHandler);
        previewStopHandler = null;
      }

      if (!clip) {
        el.previewVideo.removeAttribute('src');
        el.previewVideo.load();
        el.currentClipLabel.textContent = '선택된 클립 없음';
        el.addRangeBtn.disabled = true;
        el.resetRangeBtn.disabled = true;
        el.filmstripThumbs.innerHTML = '';
        rebuildTimelineStructure();
        renderRanges();
        renderClips();
        updateButtons();
        return;
      }

      el.previewVideo.src = clip.url;
      el.currentClipLabel.textContent = `${clip.name} (${clip.width}×${clip.height})`;
      el.addRangeBtn.disabled = false;
      el.resetRangeBtn.disabled = false;

      renderFilmstrip(clip);
      rebuildTimelineStructure();
      renderRanges();
      renderClips();
      updateButtons();
    }

    function addNewRangeAtCurrentTime() {
      const clip = getSelectedClip();
      if (!clip) return;

      const cur = Number(el.previewVideo.currentTime || 0);
      const start = Number(Math.max(0, Math.min(cur, Math.max(0, clip.duration - 0.5))).toFixed(3));
      const end = Number(Math.min(clip.duration, start + Math.min(5, Math.max(0.5, clip.duration - start))).toFixed(3));

      const newRange = { id: uid(), start, end, enabled: true };
      clip.ranges.push(newRange);
      clip.activeRangeId = newRange.id;

      rebuildTimelineStructure(true);
      renderRanges();
      renderClips();
    }

    function renderClips() {
      el.clipCount.textContent = clips.length;
      el.collapsedLabel.textContent = `클립 (${clips.length})`;
      el.clipList.innerHTML = '';

      clips.forEach((clip, idx) => {
        const li = document.createElement('li');
        li.className = `clip-item ${clip.id === selectedClipId ? 'active' : ''}`;
        li.onclick = () => selectClip(clip.id);

        const checkedCount = clip.ranges.filter(r => r.enabled).length;
        const totalSel = getCheckedDuration(clip).toFixed(2);
        li.innerHTML = `
          <div class="clip-meta">
            <div class="clip-title">${idx + 1}. ${escapeHtml(clip.name)}</div>
            <div class="clip-sub">
              체크 ${checkedCount}/${clip.ranges.length} · ${totalSel}s / ${clip.duration.toFixed(2)}s
            </div>
          </div>
          <div class="clip-actions" onclick="event.stopPropagation()">
            <button type="button" title="위로" ${idx === 0 ? 'disabled' : ''} data-act="up">▲</button>
            <button type="button" title="아래로" ${idx === clips.length - 1 ? 'disabled' : ''} data-act="down">▼</button>
            <button type="button" class="danger" title="삭제" data-act="del">✕</button>
          </div>
        `;

        li.querySelector('[data-act="up"]').onclick = () => {
          const [item] = clips.splice(idx, 1);
          clips.splice(idx - 1, 0, item);
          renderClips();
        };

        li.querySelector('[data-act="down"]').onclick = () => {
          const [item] = clips.splice(idx, 1);
          clips.splice(idx + 1, 0, item);
          renderClips();
        };

        li.querySelector('[data-act="del"]').onclick = () => {
          URL.revokeObjectURL(clip.url);
          const delIdx = clips.findIndex(c => c.id === clip.id);
          clips.splice(delIdx, 1);
          if (selectedClipId === clip.id) {
            selectClip(clips[0] ? clips[0].id : null);
          } else {
            renderClips();
          }
        };

        el.clipList.appendChild(li);
      });

      updateButtons();
    }

    function updateButtons() {
      const clip = getSelectedClip();
      const clipCheckedTasks = clip
        ? clip.ranges
            .map((range, idx) => ({ clip, range, rangeIndex: idx }))
            .filter(t => t.range.enabled)
        : [];
      const allCheckedTasks = getAllCheckedTasks();

      el.clearAllBtn.disabled = clips.length === 0;
      el.playClipSeqBtn.disabled = clipCheckedTasks.length === 0;
      el.playAllSeqBtn.disabled = allCheckedTasks.length === 0;
      el.exportClipRangesBtn.disabled = clipCheckedTasks.length === 0;
      el.exportAllRangesBtn.disabled = allCheckedTasks.length === 0;
    }

    function setBusy(busy) {
      el.clearAllBtn.disabled = busy || clips.length === 0;
      el.playClipSeqBtn.disabled = busy;
      el.playAllSeqBtn.disabled = busy;
      el.exportClipRangesBtn.disabled = busy;
      el.exportAllRangesBtn.disabled = busy;
      if (!busy) updateButtons();
    }

    async function processTasks(tasks, outputFilename) {
      if (!tasks.length) return;
      stopSequencePreview(true);
      setBusy(true);
      el.resultBox.classList.remove('show');
      el.progressBar.value = 0;
      setLog(`[작업 시작] 총 ${tasks.length}개 체크된 구간을 추출해 하나로 합칩니다...`);

      try {
        await ensureFFmpeg();
        const mode = el.encodeMode.value;
        const segmentNames = [];

        const firstClip = tasks[0].clip;
        const baseW = firstClip.width % 2 === 0 ? firstClip.width : firstClip.width - 1;
        const baseH = firstClip.height % 2 === 0 ? firstClip.height : firstClip.height - 1;

        const writtenInputs = new Map();

        for (let i = 0; i < tasks.length; i++) {
          const { clip, range, rangeIndex } = tasks[i];
          let inName = writtenInputs.get(clip.id);
          if (!inName) {
            inName = `src_${writtenInputs.size}.mp4`;
            await ffmpeg.writeFile(inName, await fetchFile(clip.file));
            writtenInputs.set(clip.id, inName);
          }

          const segName = `seg_${i}.mp4`;
          const duration = Math.max(0.05, Number((range.end - range.start).toFixed(3)));

          appendLog(`\n[${i + 1}/${tasks.length}] ${clip.name} (구간 #${rangeIndex + 1}: ${range.start.toFixed(2)}s ~ ${range.end.toFixed(2)}s)`);

          if (mode === 'copy') {
            await ffmpeg.exec([
              '-ss', String(range.start),
              '-i', inName,
              '-t', String(duration),
              '-c', 'copy',
              '-avoid_negative_ts', 'make_zero',
              segName
            ]);
          } else {
            const vf = `scale=${baseW}:${baseH}:force_original_aspect_ratio=decrease,pad=${baseW}:${baseH}:(ow-iw)/2:(oh-ih)/2,fps=30,format=yuv420p`;
            await ffmpeg.exec([
              '-ss', String(range.start),
              '-i', inName,
              '-t', String(duration),
              '-vf', vf,
              '-c:v', 'libx264',
              '-preset', 'veryfast',
              '-crf', '22',
              '-c:a', 'aac',
              '-ar', '44100',
              '-ac', '2',
              '-b:a', '128k',
              segName
            ]);
          }

          segmentNames.push(segName);
        }

        for (const inName of writtenInputs.values()) {
          await ffmpeg.deleteFile(inName);
        }

        const finalName = 'final_merged_ranges.mp4';

        if (segmentNames.length === 1) {
          const data = await ffmpeg.readFile(segmentNames[0]);
          await ffmpeg.deleteFile(segmentNames[0]);
          publishResult(data, outputFilename);
        } else {
          appendLog('\n[병합] 선택된 구간들을 순서대로 이어 붙이는 중...');
          const listContent = segmentNames.map(name => `file '${name}'`).join('\n');
          await ffmpeg.writeFile('concat_list.txt', new TextEncoder().encode(listContent));

          await ffmpeg.exec([
            '-f', 'concat',
            '-safe', '0',
            '-i', 'concat_list.txt',
            '-c', 'copy',
            '-movflags', '+faststart',
            finalName
          ]);

          const data = await ffmpeg.readFile(finalName);

          await ffmpeg.deleteFile('concat_list.txt');
          await ffmpeg.deleteFile(finalName);
          for (const seg of segmentNames) {
            await ffmpeg.deleteFile(seg);
          }

          publishResult(data, outputFilename);
        }

        el.progressBar.value = 100;
        appendLog('\n[완료] 선택 구간 병합 파일이 생성되었습니다. "저장 위치 지정 저장" 버튼으로 원하는 폴더에 저장하세요.');
      } catch (err) {
        console.error(err);
        el.engineStatus.textContent = 'FFmpeg 초기화 오류';
        const detail = err?.stack || err?.message || String(err);
        appendLog(`\n[오류 발생] ${detail}`);
        alert(`영상 처리 중 오류가 발생했습니다: ${err?.message || err}`);
      } finally {
        setBusy(false);
      }
    }

    function publishResult(uint8Array, filename) {
      if (lastOutputUrl) URL.revokeObjectURL(lastOutputUrl);
      const blob = new Blob([uint8Array.buffer], { type: 'video/mp4' });
      lastOutputBlob = blob;
      lastOutputFilename = filename;
      lastOutputUrl = URL.createObjectURL(blob);

      const mb = (blob.size / (1024 * 1024)).toFixed(2);
      el.downloadLink.href = lastOutputUrl;
      el.downloadLink.download = filename;
      el.resultInfo.textContent = `파일명: ${filename} · 크기: ${mb} MB`;
      el.resultBox.classList.add('show');
    }

    function escapeHtml(str) {
      return str.replace(/[&<>"']/g, m => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
      }[m]));
    }

      const range = getActiveRange(clip);
      if (!clip || !range) return;
      stopSequencePreview(true);
      const cur = Number((el.previewVideo.currentTime || 0).toFixed(3));
      range.start = Math.min(cur, Math.max(0, range.end - MIN_SEG_SEC));
      updateTimelinePositions();
      renderRanges();
      renderClips();
    };

    el.setEndBtn.onclick = () => {
      const clip = getSelectedClip();
      const range = getActiveRange(clip);
      if (!clip || !range) return;
      stopSequencePreview(true);
      const cur = Number((el.previewVideo.currentTime || 0).toFixed(3));
      range.end = Math.max(cur, Math.min(clip.duration, range.start + MIN_SEG_SEC));
      updateTimelinePositions();
      renderRanges();
      renderClips();
    };

    el.previewTrimBtn.onclick = () => {
      const clip = getSelectedClip();
      const range = getActiveRange(clip);
      if (!clip || !range) return;

      stopSequencePreview(false);
      if (previewStopHandler) {
        el.previewVideo.removeEventListener('timeupdate', previewStopHandler);
      }

      el.previewVideo.currentTime = range.start;
      el.previewVideo.play();

      previewStopHandler = () => {
        if (el.previewVideo.currentTime >= range.end) {
          el.previewVideo.pause();
          el.previewVideo.removeEventListener('timeupdate', previewStopHandler);
          previewStopHandler = null;
        }
      };
      el.previewVideo.addEventListener('timeupdate', previewStopHandler);
    };
